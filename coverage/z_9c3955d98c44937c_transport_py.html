1r""" 

2Quantum transport through a finite Tight-Binding device attached to 

3semi-infinite leads: the Landauer-Buttiker picture, in which the (linear, 

4zero-temperature) conductance between two leads is the transmission 

5probability at the Fermi energy, 

6 

7.. math:: 

8 

9 G = \frac{e^2}{h}\,T(E_F)\, ,\qquad 

10 T(E) = \mathrm{Tr}\left[\Gamma_{out}\,G^r\,\Gamma_{in}\,G^a\right]\, , 

11 

12(the Caroli formula) with :math:`G^r = [E + i\eta - H - \sum_l\Sigma_l]^{-1}` 

13the device's retarded Green's function, :math:`\Sigma_l` the self-energy of 

14lead :math:`l` and :math:`\Gamma_l = i(\Sigma_l - \Sigma_l^\dagger)`. 

15 

16A lead is a semi-infinite repetition of a unit cell with Hamiltonian 

17:math:`h_0`, each cell coupled to the next one (away from the device) by 

18:math:`v`; its surface cell is coupled to some device sites by 

19:math:`\tau`. Its surface Green's function comes from the decimation 

20algorithm of Lopez Sancho, Lopez Sancho and Rubio (J. Phys. F 15, 851 

21(1985)), and :math:`\Sigma = \tau\, g_s\,\tau^\dagger`. 

22 

23Example usage:: 

24 

25 # a perfect chain of 10 sites between two chain leads: T = 1 in the band 

26 tr = Transport(sys.ham) 

27 h0, v = np.zeros((1, 1)), np.ones((1, 1)) 

28 tr.add_lead(h0, v, coupling=np.ones((1, 1)), sites=[0]) 

29 tr.add_lead(h0, v, coupling=np.ones((1, 1)), sites=[9]) 

30 T = tr.transmission(np.linspace(-1.9, 1.9, 50)) 

31"""

41def surface_green( 

42 h0: ArrayLike, v: ArrayLike, energy: float, eta: float = 1e-9, tol: float = 1e-12, 

43 max_iter: int = 10000, 

44) -> NDArray[np.complex128]: 

45 r''' 

46 Get the surface Green's function of a semi-infinite lead: cells 

47 :math:`0, 1, 2, \dots`, with :math:`H_{nn} = h_0` and 

48 :math:`H_{n,n+1} = v`, :math:`H_{n+1,n} = v^\dagger`, by the 

49 Lopez Sancho-Rubio decimation (each iteration doubles the number of 

50 cells accounted for, so it converges in a few tens of steps inside the 

51 bands, as :math:`\eta\to0^+`). 

52 

53 :param h0: Square complex array. Cell Hamiltonian. 

54 :param v: Square complex array, same shape. Coupling of a cell to the next one. 

55 :param energy: Real number. Energy. 

56 :param eta: Positive real. Default value 1e-9. Broadening. 

57 :param tol: Positive real. Default value 1e-12. Convergence threshold on 

58 the norm of the renormalized couplings. 

59 :param max_iter: Positive integer. Default value 10000. 

60 

61 :returns: 

62 * **gs** -- Complex ndarray, same shape as *h0*: :math:`g_s = [E+i\eta - h_0 - \Sigma_{bulk}]^{-1}`. 

63 ''' 

64 h0 = np.atleast_2d(np.asarray(h0, dtype='c16')) 

65 v = np.atleast_2d(np.asarray(v, dtype='c16')) 

66 error_handling.lead(h0, v) 

67 error_handling.real_number(energy, 'energy') 

68 error_handling.positive_real(eta, 'eta') 

69 error_handling.positive_real(tol, 'tol') 

70 error_handling.positive_int(max_iter, 'max_iter') 

71 m = len(h0) 

72 # The first decimation step inverts E + i eta - h0: at an energy equal to 

73 # an eigenvalue of h0 (E = 0 for a plain chain) that is ~1/eta, and the 

74 # precision is lost. Regrouping n = 1, 2 or 3 cells into one (the same 

75 # lead) moves those energies: use the best-conditioned grouping. 

76 best = None 

77 for n in (1, 2, 3): 

78 hn = np.kron(np.eye(n), h0) + np.kron(np.eye(n, k=1), v) + np.kron(np.eye(n, k=-1), v.conj().T) 

79 vn = np.kron(np.eye(n, k=1 - n), v) if n > 1 else v.copy() 

80 smin = np.linalg.svd((energy + 1j * eta) * np.eye(n * m) - hn, compute_uv=False)[-1] 

81 if best is None or smin > best[0] * (1. + 1e-12): 

82 best = (smin, hn, vn) 

83 _, hn, vn = best 

84 z = (energy + 1j * eta) * np.eye(len(hn)) 

85 eps_s, eps = hn.copy(), hn.copy() 

86 alpha, beta = vn.copy(), vn.conj().T.copy() 

87 for _ in range(max_iter): 

88 g = LA.inv(z - eps) 

89 ag, bg = alpha @ g, beta @ g 

90 eps_s = eps_s + ag @ beta 

91 eps = eps + ag @ beta + bg @ alpha 

92 alpha, beta = ag @ alpha, bg @ beta 

93 if np.linalg.norm(alpha) + np.linalg.norm(beta) < tol: 

94 break 

95 else: 

96 error_handling.converged(False, 'surface_green') 

97 return LA.inv(z - eps_s)[:m, :m]

100def lead_from_kspace(ks, direction: int = 1) -> tuple[NDArray[np.complex128], NDArray[np.complex128]]: 

101 r''' 

102 Get the cell Hamiltonian :math:`h_0` and coupling :math:`v` of a lead 

103 from a 1D periodic model (e.g. a ribbon, see *kspace.ribbon*), whose 

104 hoppings reach at most the neighbouring cells. 

105 

106 :param ks: **KSpace** instance, 1D. 

107 :param direction: +1 or -1. Default value 1. Direction, along the 

108 model's primitive vector, in which the lead extends away from the 

109 device: :math:`v` gathers the hoppings to the cell at 

110 :math:`\mathbf{R} = \pm\mathbf{a}_1`. 

111 

112 :returns: 

113 * **h0** -- Complex ndarray, shape (norb, norb). 

114 * **v** -- Complex ndarray, shape (norb, norb). 

115 ''' 

116 error_handling.dim_exact(ks.dim, 1) 

117 error_handling.lead_direction(direction) 

118 h0 = np.diag(ks.onsite).astype('c16') + ks._onsite_offdiag 

119 v = np.zeros((ks.norb, ks.norb), 'c16') 

120 for i, j, (n,), t in ks._hop_cells(): 

121 error_handling.nearest_cells(n) 

122 if n == 0: 

123 h0[i, j] += t 

124 elif n == direction: 

125 v[i, j] += t 

126 return h0, v

129class Transport(): 

130 r''' 

131 A finite device, of Hamiltonian *ham* (e.g. *System.ham*), to which 

132 semi-infinite leads are attached with *add_lead*; *transmission* then 

133 gives the Landauer transmission between two of them. 

134 

135 :param ham: Square matrix (sparse or dense). Device Hamiltonian. 

136 ''' 

137 

138 def __init__(self, ham) -> None: 

139 ham = ham.toarray() if hasattr(ham, 'toarray') else np.asarray(ham) 

140 error_handling.square_matrix(ham, 'ham') 

141 self.ham = ham.astype('c16') 

142 self.leads = [] # list of (h0, v, coupling, sites) 

143 

144 def add_lead(self, h0: ArrayLike, v: ArrayLike, coupling: ArrayLike, sites: list[int]) -> None: 

145 r''' 

146 Attach a semi-infinite lead (see *surface_green*). 

147 

148 :param h0: Square complex array, shape (m, m). Lead cell Hamiltonian. 

149 :param v: Square complex array, shape (m, m). Coupling of a lead cell 

150 to the next one, away from the device. 

151 :param coupling: Complex array, shape (len(sites), m). Hoppings 

152 :math:`\tau` between the device *sites* and the lead's surface cell. 

153 :param sites: List of device site (row) indices the lead touches. 

154 ''' 

155 h0 = np.atleast_2d(np.asarray(h0, dtype='c16')) 

156 v = np.atleast_2d(np.asarray(v, dtype='c16')) 

157 error_handling.lead(h0, v) 

158 coupling = np.atleast_2d(np.asarray(coupling, dtype='c16')) 

159 error_handling.lead_coupling(coupling, sites, len(h0), len(self.ham)) 

160 self.leads.append((h0, v, coupling, list(sites))) 

161 

162 def self_energy(self, lead: int, energy: float, eta: float = 1e-9) -> NDArray[np.complex128]: 

163 r''' 

164 Get the self-energy :math:`\Sigma = \tau g_s \tau^\dagger` of a lead, 

165 embedded in the device space. 

166 

167 :param lead: Integer. Lead index (in the order of *add_lead*). 

168 :param energy: Real number. 

169 :param eta: Positive real. Default value 1e-9. 

170 

171 :returns: 

172 * **sigma** -- Complex ndarray, shape of *ham*. 

173 ''' 

174 error_handling.lead_index(lead, len(self.leads)) 

175 h0, v, tau, sites = self.leads[lead] 

176 sigma = np.zeros_like(self.ham) 

177 sigma[np.ix_(sites, sites)] = tau @ surface_green(h0, v, energy, eta) @ tau.conj().T 

178 return sigma 

179 

180 def get_green(self, energy: float, eta: float = 1e-9) -> NDArray[np.complex128]: 

181 r''' 

182 Get the retarded Green's function of the device with all its leads, 

183 :math:`G^r = [E + i\eta - H - \sum_l \Sigma_l]^{-1}`. 

184 

185 :param energy: Real number. 

186 :param eta: Positive real. Default value 1e-9. 

187 

188 :returns: 

189 * **G** -- Complex ndarray, shape of *ham*. 

190 ''' 

191 error_handling.real_number(energy, 'energy') 

192 error_handling.positive_real(eta, 'eta') 

193 sigma = sum((self.self_energy(l, energy, eta) for l in range(len(self.leads))), 

194 np.zeros_like(self.ham)) 

195 return LA.inv((energy + 1j * eta) * np.eye(len(self.ham)) - self.ham - sigma) 

196 

197 def transmission( 

198 self, energies: ArrayLike, lead_in: int = 0, lead_out: int = 1, eta: float = 1e-9, 

199 ) -> NDArray[np.float64]: 

200 r''' 

201 Get the transmission :math:`T(E) = \mathrm{Tr}[\Gamma_{out} G^r 

202 \Gamma_{in} G^a]` from lead *lead_in* to lead *lead_out*: the 

203 two-terminal conductance in units of :math:`e^2/h` (per spin, for a 

204 spinless model). 

205 

206 :param energies: Real array. Energies. 

207 :param lead_in: Integer. Default value 0. 

208 :param lead_out: Integer. Default value 1. 

209 :param eta: Positive real. Default value 1e-9. 

210 

211 :returns: 

212 * **T** -- Real ndarray, same length as *energies*. 

213 ''' 

214 error_handling.lead_index(lead_in, len(self.leads)) 

215 error_handling.lead_index(lead_out, len(self.leads)) 

216 energies = np.atleast_1d(np.asarray(energies, dtype='f8')) 

217 out = np.zeros(len(energies)) 

218 for n, e in enumerate(energies): 

219 sig_in = self.self_energy(lead_in, e, eta) 

220 sig_out = self.self_energy(lead_out, e, eta) 

221 gam_in = 1j * (sig_in - sig_in.conj().T) 

222 gam_out = 1j * (sig_out - sig_out.conj().T) 

223 g = self.get_green(e, eta) 

224 out[n] = np.trace(gam_out @ g @ gam_in @ g.conj().T).real 

225 return out
