1r""" 

2The kernel polynomial method (KPM; Silver and Roder 1994, Weisse, 

3Wellein, Alvermann and Fehske, Rev. Mod. Phys. 78, 275 (2006)): spectral 

4quantities of a large sparse Hamiltonian from its Chebyshev moments, 

5without diagonalizing it. The spectrum is first mapped into 

6:math:`(-1, 1)`, :math:`\tilde H = (H - b)/a`; then 

7 

8.. math:: 

9 

10 \delta(E - H) = \sum_{m=0}^{M-1} c_m(E)\, T_m(\tilde H)\, ,\qquad 

11 c_m(E) = \frac{(2-\delta_{m0})\,g_m\,T_m(\epsilon)}{a\pi\sqrt{1-\epsilon^2}}\, , 

12 

13with :math:`\epsilon = (E-b)/a`, :math:`T_m` the Chebyshev polynomials 

14(computed by the recursion :math:`T_{m+1}(\tilde H)|r\rangle = 

152\tilde H\,T_m(\tilde H)|r\rangle - T_{m-1}(\tilde H)|r\rangle`, one sparse 

16matrix-vector product each), and :math:`g_m` a kernel that damps the Gibbs 

17oscillations of the truncated series (Jackson: an almost Gaussian 

18broadening of width about :math:`\pi a/M`). Traces are estimated with a 

19few random-phase vectors, :math:`\mathrm{Tr}\,A \approx 

20\frac1R\sum_r\langle r|A|r\rangle`, whose error falls as :math:`1/\sqrt{RN}`: 

21the cost is :math:`O(MRN)` for :math:`N` sites, so millions of sites are 

22within reach. 

23 

24The Hamiltonian must be Hermitian (*System.ham*, a sparse matrix). 

25"""

39def spectral_bounds(ham) -> tuple[float, float]: 

40 r''' 

41 Get bounds on the spectrum of a Hermitian matrix from Gershgorin's 

42 circle theorem: every eigenvalue lies within 

43 :math:`H_{ii} \pm \sum_{j\neq i}|H_{ij}|` for some :math:`i`. 

44 

45 :param ham: Square sparse (or dense) Hermitian matrix. 

46 

47 :returns: 

48 * **e_min**, **e_max** -- Real numbers. 

49 ''' 

50 ham = sparse.csr_matrix(ham) 

51 diag = ham.diagonal().real 

52 radii = np.asarray(abs(ham).sum(axis=1)).ravel() - np.abs(ham.diagonal()) 

53 return float(np.min(diag - radii)), float(np.max(diag + radii))

56def jackson_kernel(n_moments: int) -> NDArray[np.float64]: 

57 r''' 

58 Get the Jackson kernel 

59 

60 .. math:: 

61 

62 g_m = \frac{(M-m+1)\cos\frac{\pi m}{M+1} 

63 + \sin\frac{\pi m}{M+1}\cot\frac{\pi}{M+1}}{M+1}\, , 

64 

65 the optimal choice for spectral densities (positive, and resolution 

66 about :math:`\pi/M` in the scaled energy). 

67 

68 :param n_moments: Positive integer :math:`M`. 

69 

70 :returns: 

71 * **g** -- Real ndarray, shape (n_moments,). 

72 ''' 

73 error_handling.positive_int(n_moments, 'n_moments') 

74 m = np.arange(n_moments) 

75 q = PI / (n_moments + 1) 

76 return ((n_moments - m + 1) * np.cos(q * m) + np.sin(q * m) / np.tan(q)) / (n_moments + 1)

79def lorentz_kernel(n_moments: int, lam: float = 4.) -> NDArray[np.float64]: 

80 r''' 

81 Get the Lorentz kernel :math:`g_m = \sinh[\lambda(1 - m/M)]/\sinh\lambda`, 

82 which gives Lorentzian broadening (the better choice for Green's 

83 functions). 

84 

85 :param n_moments: Positive integer :math:`M`. 

86 :param lam: Positive real. Default value 4. 

87 

88 :returns: 

89 * **g** -- Real ndarray, shape (n_moments,). 

90 ''' 

91 error_handling.positive_int(n_moments, 'n_moments') 

92 error_handling.positive_real(lam, 'lam') 

93 return np.sinh(lam * (1. - np.arange(n_moments) / n_moments)) / np.sinh(lam)

96def _setup(ham, n_moments, kernel, bounds, e_grid, n_grid=1001): 

97 ''' 

98 Private function. Validate, rescale the Hamiltonian, and build the 

99 coefficients c_m(E) (shape (n_moments, len(e_grid))). 

100 ''' 

101 ham = sparse.csr_matrix(ham, dtype='c16') 

102 error_handling.square_matrix(ham, 'ham') 

103 error_handling.hermitian(ham) 

104 error_handling.positive_int(n_moments, 'n_moments') 

105 error_handling.kpm_kernel(kernel) 

106 if bounds is None: 

107 bounds = spectral_bounds(ham) 

108 error_handling.lims(bounds) 

109 e_min, e_max = bounds 

110 a = (e_max - e_min) / (2. - 0.02) # keep the scaled spectrum inside (-0.99, 0.99) 

111 b = (e_max + e_min) / 2. 

112 if e_grid is None: 

113 e_grid = b + a * np.cos(PI * (np.arange(n_grid) + 0.5) / n_grid)[::-1] 

114 e_grid = np.atleast_1d(np.asarray(e_grid, dtype='f8')) 

115 error_handling.ndarray_empty(e_grid, 'e_grid') 

116 eps = (e_grid - b) / a 

117 error_handling.kpm_grid(eps) 

118 g = jackson_kernel(n_moments) if kernel == 'jackson' else lorentz_kernel(n_moments) 

119 tm = np.cos(np.arange(n_moments)[:, None] * np.arccos(eps)[None, :]) 

120 weight = np.where(np.arange(n_moments) == 0, 1., 2.) 

121 coef = (weight * g)[:, None] * tm / (a * PI * np.sqrt(1. - eps ** 2))[None, :] 

122 ham_s = (ham - b * sparse.identity(ham.shape[0], format='csr')) / a 

123 return ham_s, e_grid, coef, a, b

126def _vectors(n: int, n_random: int | None, seed) -> NDArray[np.complex128]: 

127 ''' 

128 Private function. Random-phase vectors (columns, |r_i| = 1), or, if 

129 n_random is None, the full basis (the exact trace). 

130 ''' 

131 if n_random is None: 

132 return np.eye(n, dtype='c16') 

133 error_handling.positive_int(n_random, 'n_random') 

134 rng = np.random.default_rng(seed) 

135 return np.exp(2j * PI * rng.random((n, n_random)))

138def _chebyshev(ham_s, vec: NDArray[np.complex128], n_moments: int): 

139 ''' 

140 Private function. Yield T_m(H~) vec for m < n_moments, keeping only two 

141 vectors in memory. 

142 ''' 

143 prev, cur = None, vec 

144 for m in range(n_moments): 

145 yield cur 

146 prev, cur = cur, (ham_s @ cur if m == 0 else 2 * (ham_s @ cur) - prev)

149def dos( 

150 ham, n_moments: int = 256, n_random: int | None = 10, e_grid: ArrayLike | None = None, 

151 kernel: str = 'jackson', seed=None, bounds: tuple[float, float] | None = None, 

152) -> tuple[NDArray[np.float64], NDArray[np.float64]]: 

153 r''' 

154 Get the density of states :math:`\rho(E) = \mathrm{Tr}\,\delta(E - H)` 

155 (normalized, like *tbkit.dos.density_of_states*, to integrate to the 

156 number of states). 

157 

158 :param ham: Square sparse Hermitian matrix (e.g. *System.ham*). 

159 :param n_moments: Positive integer. Default value 256. Number of 

160 Chebyshev moments (the energy resolution is about 

161 :math:`\pi a/M`, :math:`2a` the spectral width). 

162 :param n_random: Positive integer, or None. Default value 10. Number of 

163 random vectors for the trace; None for the exact trace (N vectors: 

164 small systems only). 

165 :param e_grid: Real array. Default value None: 1001 Chebyshev nodes over 

166 the spectrum. Energies, inside the spectral bounds. 

167 :param kernel: String. Default value 'jackson'. 'jackson' or 'lorentz'. 

168 :param seed: Default value None. Seed of the random vectors. 

169 :param bounds: Pair of reals. Default value None (Gershgorin bounds, see 

170 *spectral_bounds*). Bounds enclosing the whole spectrum. 

171 

172 :returns: 

173 * **e_grid** -- Real ndarray. 

174 * **rho** -- Real ndarray, same shape. 

175 ''' 

176 ham_s, e_grid, coef, _, _ = _setup(ham, n_moments, kernel, bounds, e_grid) 

177 vec = _vectors(ham_s.shape[0], n_random, seed) 

178 mu = np.array([np.vdot(vec, t).real for t in _chebyshev(ham_s, vec, n_moments)]) 

179 if n_random is not None: 

180 mu = mu / n_random # E<r|A|r> = Tr A for random phases 

181 return e_grid, mu @ coef

184def dos_from_levels( 

185 levels: ArrayLike, n_moments: int, bounds: tuple[float, float], e_grid: ArrayLike, 

186 kernel: str = 'jackson', 

187) -> NDArray[np.float64]: 

188 r''' 

189 Get the KPM density of states of a *known* spectrum: every level 

190 broadened by exactly the truncated, kernel-damped Chebyshev series that 

191 *dos* uses. It is the reference *dos* converges to (up to the random 

192 trace), e.g. for the bands of a periodic model on a k-mesh. 

193 

194 :param levels: Real array. Energy levels. 

195 :param n_moments: Positive integer. Number of Chebyshev moments. 

196 :param bounds: Pair of reals. Spectral bounds (those of *dos*, e.g. 

197 *spectral_bounds* of the Hamiltonian it is compared with). 

198 :param e_grid: Real array. Energies. 

199 :param kernel: String. Default value 'jackson'. 

200 

201 :returns: 

202 * **rho** -- Real ndarray, same shape as *e_grid* (integrating to the number of levels). 

203 ''' 

204 levels = np.asarray(levels, dtype='f8').ravel() 

205 error_handling.ndarray_empty(levels, 'levels') 

206 _, e_grid, coef, a, b = _setup(sparse.identity(1, format='csr'), n_moments, kernel, bounds, e_grid) 

207 error_handling.kpm_grid((levels - b) / a) 

208 tm = np.cos(np.arange(n_moments)[:, None] * np.arccos((levels - b) / a)[None, :]) 

209 return coef.T @ tm.sum(axis=1)

212def ldos( 

213 ham, site: int, n_moments: int = 256, e_grid: ArrayLike | None = None, 

214 kernel: str = 'jackson', bounds: tuple[float, float] | None = None, 

215) -> tuple[NDArray[np.float64], NDArray[np.float64]]: 

216 r''' 

217 Get the local density of states on one site (or orbital), 

218 :math:`\rho_i(E) = \langle i|\delta(E-H)|i\rangle`, exactly (no random vectors). 

219 

220 :param ham: Square sparse Hermitian matrix. 

221 :param site: Integer. Site (row) index. 

222 :param n_moments: Positive integer. Default value 256. 

223 :param e_grid: See *dos*. 

224 :param kernel: See *dos*. 

225 :param bounds: See *dos*. 

226 

227 :returns: 

228 * **e_grid** -- Real ndarray. 

229 * **rho** -- Real ndarray, same shape. 

230 ''' 

231 ham_s, e_grid, coef, _, _ = _setup(ham, n_moments, kernel, bounds, e_grid) 

232 error_handling.site_index(site, ham_s.shape[0]) 

233 vec = np.zeros(ham_s.shape[0], 'c16') 

234 vec[site] = 1. 

235 mu = np.array([t[site].real for t in _chebyshev(ham_s, vec, n_moments)]) 

236 return e_grid, mu @ coef

239def conductivity( 

240 ham, x: ArrayLike, n_moments: int = 128, n_random: int | None = 10, 

241 e_grid: ArrayLike | None = None, kernel: str = 'jackson', seed=None, 

242 bounds: tuple[float, float] | None = None, area: float | None = None, 

243) -> tuple[NDArray[np.float64], NDArray[np.float64]]: 

244 r''' 

245 Get the zero-temperature DC conductivity along :math:`x` from the 

246 Kubo-Greenwood formula, 

247 

248 .. math:: 

249 

250 \sigma_{xx}(E) = \frac{\pi}{\Omega}\, 

251 \mathrm{Tr}\left[v_x\,\delta(E-H)\,v_x\,\delta(E-H)\right]\, ,\qquad 

252 v_x = i[H, x]\, , 

253 

254 in units of :math:`e^2/\hbar` (:math:`\hbar = e = 1`), by the 

255 two-dimensional Chebyshev expansion of Weisse et al. (the moments 

256 :math:`\mu_{mn} = \mathrm{Tr}[v_x T_m(\tilde H) v_x T_n(\tilde H)]`, 

257 :math:`O(M N)` memory, the DOS needing only :math:`O(N)`). 

258 

259 :param ham: Square sparse Hermitian matrix. 

260 :param x: Real array, one coordinate per site (e.g. *lat.coor['x']*). 

261 :param n_moments: Positive integer. Default value 128. 

262 :param n_random: Positive integer, or None. Default value 10. See *dos*. 

263 :param e_grid: See *dos*. 

264 :param kernel: See *dos*. 

265 :param seed: See *dos*. 

266 :param bounds: See *dos*. 

267 :param area: Positive real. Default value None (the number of sites): 

268 the area (volume) :math:`\Omega` of the sample. 

269 

270 :returns: 

271 * **e_grid** -- Real ndarray. 

272 * **sigma** -- Real ndarray, same shape. 

273 ''' 

274 ham_s, e_grid, coef, a, b = _setup(ham, n_moments, kernel, bounds, e_grid) 

275 n = ham_s.shape[0] 

276 x = np.asarray(x, dtype='f8') 

277 error_handling.ndarray(x, 'x', n) 

278 if area is None: 

279 area = float(n) 

280 error_handling.positive_real(area, 'area') 

281 # v = i [H, x]: v_ij = i H_ij (x_j - x_i); the scale a comes back below 

282 ham_c = sparse.csr_matrix(ham, dtype='c16') 

283 vel = 1j * (ham_c @ sparse.diags(x) - sparse.diags(x) @ ham_c) 

284 vec = _vectors(n, n_random, seed) 

285 mu = np.zeros((n_moments, n_moments), 'c16') 

286 for r in vec.T: 

287 left = np.array(list(_chebyshev(ham_s, vel @ r, n_moments))) # T_m v |r> 

288 right = np.array([vel @ t for t in _chebyshev(ham_s, r, n_moments)]) # v T_n |r> 

289 mu += left.conj() @ right.T 

290 if n_random is not None: 

291 mu = mu / n_random 

292 sigma = PI / area * np.einsum('me,mn,ne->e', coef, mu, coef).real 

293 return e_grid, sigma

296def hall_conductivity( 

297 ham, vx, vy, n_moments: int = 256, n_random: int | None = 10, 

298 e_grid: ArrayLike | None = None, kernel: str = 'jackson', seed=None, 

299 bounds: tuple[float, float] | None = None, area: float | None = None, 

300 temperature: float = 0., 

301) -> tuple[NDArray[np.float64], NDArray[np.float64]]: 

302 r''' 

303 Get the Hall conductivity :math:`\sigma_{xy}(\mu)` of a large sample, at 

304 any Fermi level and temperature, from the Kubo-Bastin formula expanded 

305 in Chebyshev polynomials (Garcia, Covaci and Rappoport, Phys. Rev. Lett. 

306 114, 116602 (2015), Eqs. (1)-(5)). Bastin's formula, 

307 

308 .. math:: 

309 

310 \sigma^{B}_{\alpha\beta}(\mu) = \frac{ie^2\hbar}{\Omega}\int d\varepsilon\,f(\varepsilon)\, 

311 \mathrm{Tr}\left[v_\alpha\,\delta(\varepsilon-H)\,v_\beta\,\frac{dG^+}{d\varepsilon} 

312 - v_\alpha\,\frac{dG^-}{d\varepsilon}\,v_\beta\,\delta(\varepsilon-H)\right]\, , 

313 

314 :math:`G^\pm = (\varepsilon - H \pm i0)^{-1}`, becomes, with the spectrum 

315 rescaled into :math:`(-1, 1)`, :math:`\tilde H = (H-b)/a`, 

316 

317 .. math:: 

318 

319 \sigma^{B}_{\alpha\beta}(\mu) = \frac{4e^2\hbar}{\pi\Omega a^2}\int_{-1}^{1} 

320 d\tilde\varepsilon\,\frac{f(\tilde\varepsilon)}{(1-\tilde\varepsilon^2)^2} 

321 \sum_{m,n}\Gamma_{nm}(\tilde\varepsilon)\,\mu^{\alpha\beta}_{nm}\, , 

322 

323 .. math:: 

324 

325 \mu^{\alpha\beta}_{mn} = \frac{g_mg_n}{(1+\delta_{m0})(1+\delta_{n0})} 

326 \mathrm{Tr}\left[v_\alpha T_m(\tilde H)\,v_\beta T_n(\tilde H)\right]\, , 

327 

328 .. math:: 

329 

330 \Gamma_{mn}(\tilde\varepsilon) = (\tilde\varepsilon - in\sqrt{1-\tilde\varepsilon^2})\, 

331 e^{in\theta}\,T_m(\tilde\varepsilon) 

332 + (\tilde\varepsilon + im\sqrt{1-\tilde\varepsilon^2})\,e^{-im\theta}\,T_n(\tilde\varepsilon)\, , 

333 

334 :math:`\theta = \arccos\tilde\varepsilon`, :math:`g_m` the kernel. The 

335 moments :math:`\mu_{mn}` (the expensive part: :math:`5MR` sparse 

336 products, :math:`O(MN)` memory) serve every Fermi level and temperature. 

337 The energy integral runs over :math:`|\tilde\varepsilon|\le 0.995` 

338 (the spectrum lies within :math:`\pm0.99`): at the ends, 

339 :math:`(1-\tilde\varepsilon^2)^{-2}` would only amplify rounding errors. 

340 

341 **Hall part.** The value is computed from the antisymmetric moments 

342 :math:`(\mu^{xy}_{mn} - \mu^{yx}_{mn})/2`, i.e. it is the Hall 

343 conductivity proper, :math:`(\sigma^B_{xy} - \sigma^B_{yx})/2`, the part 

344 that the Berry curvature gives (*KSpace.hall_conductivity*). The 

345 symmetric part :math:`(\sigma^B_{xy} + \sigma^B_{yx})/2` is a 

346 Fermi-surface term, like :math:`\sigma_{xx}` (see *conductivity*): it 

347 vanishes by symmetry in many lattices (e.g. with a :math:`C_3` axis), 

348 but in a clean metal of lower symmetry it grows as the kernel's 

349 broadening :math:`\pi a/M` shrinks (the Drude weight), and with random 

350 vectors it is the main source of noise. Leaving it out removes that 

351 noise too. In a clean metal the random-vector estimate is still noisy 

352 inside the bands (increase *n_random*, or the size of the sample); 

353 disorder, which makes the Fermi-surface terms finite, and gaps, where 

354 they vanish, are much quieter. 

355 

356 **Sign and units.** The value returned is :math:`-(\sigma^B_{xy}-\sigma^B_{yx})/2` in units 

357 of :math:`e^2/h` (:math:`\hbar = e = 1`), the sign of 

358 *KSpace.hall_conductivity*: :math:`C` in the gap of a Chern insulator 

359 of Chern number :math:`C` (*KSpace.chern_number*). Bastin's 

360 :math:`\sigma^B_{xy}` is the Ohm's-law tensor, :math:`j_x = \sigma^B_{xy}E_y` 

361 (see *KSpace.hall_conductivity*). 

362 

363 **Sample.** The trace must run over a sample without edges -- a torus 

364 (*KSpace.finite_ham* and *KSpace.finite_velocity* with 

365 ``periodic=True``): over a whole open flake, the edge currents cancel 

366 the bulk and the trace vanishes in a gap. 

367 

368 :param ham: Square sparse Hermitian matrix. 

369 :param vx: Square sparse (or dense) Hermitian matrix, like *ham*. Velocity 

370 :math:`v_x = i[H, x]` (see *KSpace.finite_velocity*). For a spin 

371 Hall conductivity, pass the spin current :math:`\{s, v_x\}/2` 

372 instead (units :math:`e/2\pi`, see *KSpace.spin_hall_conductivity*). 

373 :param vy: Square sparse (or dense) Hermitian matrix. Velocity :math:`v_y`. 

374 :param n_moments: Positive integer. Default value 256. Number of 

375 Chebyshev moments :math:`M` (energy resolution about :math:`\pi a/M`). 

376 :param n_random: Positive integer, or None. Default value 10. See *dos*. 

377 :param e_grid: Real array. Default value None: 1001 Chebyshev nodes over 

378 the spectrum. Fermi energies :math:`\mu`. 

379 :param kernel: String. Default value 'jackson'. See *dos*. 

380 :param seed: See *dos*. 

381 :param bounds: See *dos*. 

382 :param area: Positive real. Default value None (the number of orbitals). 

383 Area :math:`\Omega` of the sample (e.g. number of cells times the 

384 cell area). 

385 :param temperature: Positive real or zero. Default value 0. In energy units. 

386 

387 :returns: 

388 * **e_grid** -- Real ndarray. 

389 * **sigma** -- Real ndarray, same shape, in units of :math:`e^2/h`. 

390 ''' 

391 ham_s, e_grid, _, a, b = _setup(ham, n_moments, kernel, bounds, e_grid) 

392 n = ham_s.shape[0] 

393 error_handling.velocity(vx, n, 'vx') 

394 error_handling.velocity(vy, n, 'vy') 

395 vx = sparse.csr_matrix(vx, dtype='c16') 

396 vy = sparse.csr_matrix(vy, dtype='c16') 

397 error_handling.hermitian(vx) 

398 error_handling.hermitian(vy) 

399 if area is None: 

400 area = float(n) 

401 error_handling.positive_real(area, 'area') 

402 error_handling.positive_real_zero(temperature, 'temperature') 

403 g = jackson_kernel(n_moments) if kernel == 'jackson' else lorentz_kernel(n_moments) 

404 w = g / np.where(np.arange(n_moments) == 0, 2., 1.) 

405 mu = np.zeros((n_moments, n_moments), 'c16') 

406 for r in _vectors(n, n_random, seed).T: 

407 # <r| v_x T_m = (T_m v_x |r>)^dagger and v_y T_n |r>, for mu^xy; 

408 # the same with x and y swapped, for mu^yx 

409 t_r = list(_chebyshev(ham_s, r, n_moments)) 

410 left_x = np.array(list(_chebyshev(ham_s, vx @ r, n_moments))) 

411 left_y = np.array(list(_chebyshev(ham_s, vy @ r, n_moments))) 

412 right_y = np.array([vy @ t for t in t_r]) 

413 right_x = np.array([vx @ t for t in t_r]) 

414 mu += (left_x.conj() @ right_y.T - left_y.conj() @ right_x.T) / 2 

415 if n_random is not None: 

416 mu = mu / n_random 

417 mu = w[:, None] * mu * w[None, :] 

418 # the energy integrand on a grid uniform in theta = arccos(eps) 

419 n_int = max(2001, 8 * n_moments + 1) 

420 theta = np.linspace(np.arccos(-0.995), np.arccos(0.995), n_int) 

421 eps = np.cos(theta) 

422 root = np.sin(theta) 

423 m = np.arange(n_moments)[:, None] 

424 cheb = np.cos(m * theta[None, :]) 

425 plus = (eps[None, :] - 1j * m * root[None, :]) * np.exp(1j * m * theta[None, :]) 

426 minus = (eps[None, :] + 1j * m * root[None, :]) * np.exp(-1j * m * theta[None, :]) 

427 # sum_{n,m} Gamma_nm mu_nm, Gamma_nm = plus_m T_n + minus_n T_m 

428 gamma_mu = np.sum(cheb * (mu @ plus), axis=0) + np.sum(minus * (mu @ cheb), axis=0) 

429 integrand = (gamma_mu / (1. - eps ** 2) ** 2).real 

430 prefactor = -2 * PI * 4. / (PI * area * a ** 2) # e^2/hbar = 2 pi e^2/h, TKNN sign 

431 energies = a * eps + b 

432 if temperature == 0: 

433 cum = np.concatenate([[0.], np.cumsum(0.5 * (integrand[1:] + integrand[:-1]) * np.diff(eps))]) 

434 sigma = prefactor * np.interp(e_grid, energies, cum) 

435 else: 

436 f = np.array([occupation.fermi_dirac(energies, float(e), temperature) for e in e_grid]) 

437 sigma = prefactor * np.trapezoid(f * integrand[None, :], eps, axis=1) 

438 return e_grid, sigma
